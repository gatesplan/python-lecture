def bfs(graph, start): visited = {node: False for node in graph} queue = deque([start]) visited[start] = True result = [] while queue: v = queue.popleft() result.append(v) for neighbor in sorted(graph[v]): if not visited[neighbor]: visited[neighbor] = True queue.append(neighbor) return result
실행 결과
[1, 2, 3, 4, 5, 6, 7]

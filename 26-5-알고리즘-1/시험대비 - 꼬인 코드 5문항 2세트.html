def comb(n, r): d = [[1] * (n + 1)] * (n + 1) for i in range(1, n + 1): for j in range(1, i): d[i][j] = d[i - 1][j - 1] + d[i - 1][j] return d[n][r], d[0] print(comb(3, 1)) print(comb(4, 2))

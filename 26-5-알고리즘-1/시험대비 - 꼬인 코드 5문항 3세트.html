def find(arr, target): low, high = 0, len(arr) - 1 while low <= high: mid = (low + high) // 2 if arr[mid] == target: return mid if arr[mid] < target: low = mid + 1 else: high = mid - 1 return -1 data = [7, 2, 9, 4, 1, 8, 5] print(find(data, 4), find(data, 7), find(data, 9)) print(find(data, 8), find(data, 5)) print(find(sorted(data), 7), data.index(7))

def ways(n, steps, memo={}): if n == 0: return 1 if n < 0: return 0 if n not in memo: total = 0 for s in steps: total += ways(n - s, steps) memo[n] = total return memo[n] print(ways(4, [1, 2])) print(ways(4, [1, 3])) print(ways(5, [1, 3])) print(ways(4, [1, 3], {}))
